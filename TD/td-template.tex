% -- Police du corps de texte alignée sur celle du titre (Latin Modern Sans) --
% Le titre est composé en \sffamily ; on l'applique aussi au corps du document
% pour une identité typographique cohérente (mainfont ne pilote que le \rmfamily).
\renewcommand{\familydefault}{\sfdefault}

% -- Titres de section colorés --
% Pandoc mappe le "##" du .qmd (plus haut niveau de titre utilisé dans le
% document) sur \section, et "###" sur \subsection.
% Numérotation forcée à "Ex. N" quel que soit le réglage number-sections du
% .qmd (secnumdepth=1 : seul \section est numéroté).
\setcounter{secnumdepth}{1}

\titleformat{\section}
  {\color{tdnavy}\normalfont\Large\bfseries}{Exercice \thesection : }{0.5em}{}

% -- Titres "### Solution" : texte "Solution de l'Ex. N" et filet bleu --
% Le contenu tapé après "### " dans le .qmd est ignoré : le titre affiché est
% entièrement recalculé à partir du numéro d'exercice courant (\thesection).
\titleformat{\subsection}
  {\color{tdblue}\normalfont\normalsize\bfseries}
  {}{0pt}
  {\color{tdblue}\rule{0.3\linewidth}{1.2pt}\\[6pt]}
\let\tdoldsubsection\subsection
\renewcommand{\subsection}[1]{\tdoldsubsection{\tdtab~Solution de l'exercice~\thesection~ :}}

% -- Contenu de l'en-tête et du pied de page --
% Par défaut, l'en-tête reprend le titre/sous-titre ; on peut le personnaliser
% indépendamment via les champs de métadonnées header-left / header-right /
% footer-left / footer-right dans l'en-tête YAML du .qmd, par exemple :
%   header-left: "..."
%   header-right: "PC — IPESUP"
%   footer-left: "Année 2026-2027"
%   footer-right: "À rendre le ..."
% (le numéro de page reste toujours affiché à droite du pied de page)
\newcommand{\tdheaderleft}{$if(header-left)$$header-left$$else$$title$$endif$}
\newcommand{\tdheaderright}{$if(header-right)$$header-right$$else$$subtitle$$endif$}
\newcommand{\tdfooterleft}{$if(footer-left)$$footer-left$$else$$subtitle$$endif$}
\newcommand{\tdfooterright}{$if(footer-right)$$footer-right$\ ---\ $endif$\thepage/\pageref{LastPage}}
